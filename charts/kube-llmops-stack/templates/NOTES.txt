1. CHECK DEPLOYMENT STATUS:
   kubectl get pods -l app.kubernetes.io/part-of=kube-llmops --watch

   Wait until all pods show "Running" or "Completed" status.
{{- if .Values.vllm.enabled }}
   Note: vLLM model loading may take 3-10 minutes depending on model size.
{{- end }}

2. ACCESS THE UIs:
{{- $host := .Values.ingress.host }}
{{- if $host }}
   Add to /etc/hosts: <NODE_IP> litellm.{{ $host }} grafana.{{ $host }} langfuse.{{ $host }} dify.{{ $host }} keycloak.{{ $host }}

{{- if .Values.litellm.enabled }}
   LiteLLM Gateway:  http://litellm.{{ $host }}
{{- end }}
{{- if .Values.observability.grafana.enabled }}
   Grafana:           http://grafana.{{ $host }}
{{- end }}
{{- if .Values.langfuse.enabled }}
   Langfuse:          http://langfuse.{{ $host }}
{{- end }}
{{- if index .Values "dify" "enabled" }}
   Dify RAG:          http://dify.{{ $host }}
{{- end }}
{{- if .Values.keycloak.enabled }}
   Keycloak SSO:      http://keycloak.{{ $host }}
{{- end }}
{{- else }}
   kubectl port-forward svc/{{ .Release.Name }}-litellm 4000:4000 &
   kubectl port-forward svc/{{ .Release.Name }}-grafana 3000:3000 &
   kubectl port-forward svc/{{ .Release.Name }}-langfuse 3001:3000 &
{{- end }}

3. RETRIEVE CREDENTIALS:
{{- if .Values.litellm.enabled }}
   LiteLLM Master Key:
     kubectl get secret {{ .Release.Name }}-litellm-pg-secret -o jsonpath='{.data.password}' | base64 -d
{{- end }}
{{- if .Values.observability.grafana.enabled }}
   Grafana Admin:
     Username: admin
     Password: (configured in values)
{{- end }}

4. QUICK VERIFICATION:
{{- if .Values.litellm.enabled }}
   # Test LLM inference
   curl -s http://localhost:4000/v1/chat/completions \
     -H "Authorization: Bearer <MASTER_KEY>" \
     -H "Content-Type: application/json" \
     -d '{"model":"{{ (first .Values.litellm.models).name | default "qwen2-5-0-5b" }}","messages":[{"role":"user","content":"Hello"}],"max_tokens":10}'
{{- end }}
